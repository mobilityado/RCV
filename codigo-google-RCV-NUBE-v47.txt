function doGet(e) {
  e = e || { parameter: {} };
  const p = e.parameter || {};
  const accion = String(p.accion || '').trim().toLowerCase();
  const callback = String(p.callback || '').trim();
  let resultado;

  if (!accion) {
    resultado = {
      ok: true,
      mensaje: 'API REPORT.IA RCV activa',
      acciones: ['usuarios', 'login', 'cloud_data', 'cloud_meta', 'notas', 'workflow', 'publication_history', 'cloud_months', 'backup_url', 'v34_module', 'v34_snapshot', 'v34_history', 'v34_general', 'v34_sessions', 'v34_heartbeat']
    };
    return responder(resultado, callback);
  }

  if (accion === 'usuarios') resultado = obtenerUsuariosDatos();
  else if (accion === 'login') resultado = validarLoginDatos(p.usuario, p.contrasena);
  else if (accion === 'cloud_meta') resultado = obtenerMetaNube(p.token, p.periodIndex);
  else if (accion === 'cloud_data') resultado = obtenerDatosNube(p.token, p.periodIndex);
  else if (accion === 'notas') resultado = obtenerNotas(p.token, p.gerencia, p.snapshotId);
  else if (accion === 'workflow') resultado = obtenerWorkflow(p.token, p.gerencia, p.snapshotId, p.periodIndex);
  else if (accion === 'publication_history') resultado = obtenerHistorialPublicaciones(p.token);
  else if (accion === 'cloud_months') resultado = obtenerMesesDisponibles(p.token);
  else if (accion === 'backup_url') resultado = obtenerUrlRespaldo(p.token);
  else if (accion === 'v34_module') resultado = v34ObtenerModulo(p.token, p.module);
  else if (accion === 'v34_snapshot') resultado = v34ObtenerSnapshot(p.token, p.module, p.snapshotId);
  else if (accion === 'v34_history') resultado = v34Historial(p.token, p.module);
  else if (accion === 'v34_general') resultado = v34General(p.token);
  else if (accion === 'v34_sessions') resultado = v34Sesiones(p.token);
  else if (accion === 'v34_heartbeat') resultado = v34Heartbeat(p.token, p.sessionId);
  else if (accion === 'v38_notifications') resultado = v38Notificaciones(p.token);
  else if (accion === 'v38_thread') resultado = v38Hilo(p.token, p.module, p.snapshotId, p.region, p.hierarchy);
  else if (accion === 'v42_thread_status') resultado = v42EstadoHilo(p.token, p.module, p.snapshotId, p.region, p.hierarchy);
  else if (accion === 'v42_notifications') resultado = v42Notificaciones(p.token);
  else if (accion === 'v46_commitment') resultado = v46ObtenerCompromiso(p.token,p.module,p.snapshotId,p.region,p.hierarchy);
  else if (accion === 'v46_commitments') resultado = v46Compromisos(p.token);
  else if (accion === 'v47_timeline') resultado = v47Timeline(p.token,p.module,p.snapshotId,p.region,p.hierarchy);
  else resultado = { ok: false, mensaje: 'Acción no válida.' };

  return responder(resultado, callback);
}

function doPost(e) {
  e = e || { parameter: {}, postData: { contents: '' } };
  const p = e.parameter || {};
  const accion = String(p.accion || '').trim().toLowerCase();
  let resultado;

  try {
    if (accion === 'subir_nube') {
      resultado = subirModeloNube(p.token, p.payload);
    } else if (accion === 'agregar_nota') {
      resultado = agregarNota(p.token, p.snapshotId, p.gerencia, p.indicador, p.texto, p.parentId, p.tipoNota);
    } else if (accion === 'actualizar_estado') {
      resultado = actualizarEstadoIncidencia(p.token, p.snapshotId, p.gerencia, p.estado, p.prioridad);
    } else if (accion === 'sincronizar_incidencias') {
      resultado = sincronizarIncidencias(p.token, p.snapshotId, p.items);
    } else if (accion === 'v34_upload') {
      resultado = v34SubirModulo(p.token, p.module, p.payload);
    } else if (accion === 'v34_plan') {
      resultado = v34GuardarPlan(p.token, p.module, p.snapshotId, p.region, p.hierarchy, p.texto);
    } else if (accion === 'v34_logout') {
      resultado = v34Logout(p.token, p.sessionId);
    } else if (accion === 'v38_note') {
      resultado = v38GuardarNota(p.token, p.module, p.snapshotId, p.region, p.hierarchy, p.texto);
    } else if (accion === 'v42_set_status') {
      resultado = v42ActualizarEstado(p.token, p.module, p.snapshotId, p.region, p.hierarchy, p.estado);
    } else if (accion === 'v38_mark_read') {
      resultado = v38MarcarLeidas(p.token, p.ids);
    } else if (accion === 'v46_set_commitment') {
      resultado = v46GuardarCompromiso(p.token,p.module,p.snapshotId,p.region,p.hierarchy,p.responsable,p.fechaCompromiso);
    } else if (accion === 'v47_set_progress') {
      resultado = v47GuardarAvance(p.token,p.module,p.snapshotId,p.region,p.hierarchy,p.avance);
    } else {
      resultado = { ok: false, mensaje: 'Acción POST no válida.' };
    }
  } catch (error) {
    resultado = { ok: false, mensaje: String(error) };
  }

  return ContentService.createTextOutput(JSON.stringify(resultado)).setMimeType(ContentService.MimeType.JSON);
}

function obtenerUsuariosDatos() {
  try {
    const ss = SpreadsheetApp.openById(ID_SHEET);
    const sh = ss.getSheets()[0];
    const values = sh.getDataRange().getDisplayValues();
    const usuarios = [];

    for (let i = 1; i < values.length; i++) {
      const usuario = String(values[i][0] || '').trim();
      const tipo = String(values[i][2] || 'USUARIO').trim().toUpperCase();
      if (usuario) usuarios.push({ usuario: usuario, tipo: tipo });
    }
    return { ok: true, usuarios: usuarios };
  } catch (error) {
    return { ok: false, mensaje: String(error) };
  }
}

function validarLoginDatos(usuario, contrasena) {
  try {
    const ss = SpreadsheetApp.openById(ID_SHEET);
    const sh = ss.getSheets()[0];
    const values = sh.getDataRange().getDisplayValues();
    const u = String(usuario || '').trim().toUpperCase();
    const p = String(contrasena || '').trim();

    for (let i = 1; i < values.length; i++) {
      const user = String(values[i][0] || '').trim().toUpperCase();
      const pass = String(values[i][1] || '').trim();
      const tipo = String(values[i][2] || 'USUARIO').trim().toUpperCase();
      if (user === u && pass === p) {
        const token = Utilities.getUuid() + '-' + Utilities.getUuid();
        const region = String(values[i][3] || (tipo === 'ADMINISTRADOR' ? 'TODAS LAS REGIONES' : tipo)).trim().toUpperCase();
        const sessionId = Utilities.getUuid();
        CacheService.getScriptCache().put('tk_' + token, JSON.stringify({
          usuario: values[i][0], tipo: tipo, region: region, sessionId: sessionId, creado: new Date().toISOString()
        }), TOKEN_TTL);
        try { v34RegistrarSesion(sessionId, values[i][0], tipo, region); } catch (_) {}
        return { ok: true, usuario: values[i][0], tipo: tipo, region: region, sessionId: sessionId, token: token };
      }
    }
    return { ok: false, mensaje: 'Usuario o contraseña incorrectos.' };
  } catch (error) {
    return { ok: false, mensaje: String(error) };
  }
}

function sesionPorToken(token) {
  const t = String(token || '').trim();
  if (!t) return null;
  const raw = CacheService.getScriptCache().get('tk_' + t);
  if (!raw) return null;
  try { return JSON.parse(raw); } catch (_) { return null; }
}

function exigirSesion(token) {
  const s = sesionPorToken(token);
  if (!s) throw new Error('Sesión vencida. Cierra sesión e ingresa nuevamente.');
  return s;
}


function indiceMesSeguro(periodIndex) {
  const n = Number(periodIndex);
  if (isFinite(n) && n >= 0 && n <= 11) return Math.floor(n);
  return null;
}

function nombreHojaMes(periodIndex) {
  const i = indiceMesSeguro(periodIndex);
  if (i === null) throw new Error('Periodo mensual no válido.');
  return 'REPORTIA_' + MESES_NUBE[i];
}

function asegurarHojaMes(periodIndex) {
  const ss = SpreadsheetApp.openById(ID_SHEET_NUBE);
  const name = nombreHojaMes(periodIndex);
  let sh = ss.getSheetByName(name);
  if (!sh) sh = ss.insertSheet(name);
  if (sh.getLastRow() === 0) {
    sh.getRange(1,1,1,9).setValues([['SNAPSHOT_ID','FECHA','USUARIO','TIPO','INDICE_CHUNK','TOTAL_CHUNKS','PERIODO','ANIO','CONTENIDO']]);
    sh.setFrozenRows(1);
  }
  return sh;
}

function extraerAnioPayload(payload) {
  const candidates = [
    payload && payload.model && payload.model.month,
    payload && payload.model && payload.model.period,
    payload && payload.periodLabel
  ].filter(Boolean).join(' ');
  const m = String(candidates).match(/\b(20\d{2})\b/);
  return m ? m[1] : String(new Date().getFullYear());
}

function leerPayloadMes(periodIndex) {
  const i = indiceMesSeguro(periodIndex);
  if (i === null) return null;
  const sh = asegurarHojaMes(i);
  if (sh.getLastRow() < 2) return null;
  const values = sh.getRange(2,1,sh.getLastRow()-1,9).getValues();
  if (!values.length) return null;
  values.sort((a,b)=>Number(a[4])-Number(b[4]));
  const json = values.map(r=>String(r[8] || '')).join('');
  if (!json) return null;
  return JSON.parse(json);
}

function leerPayloadNubeMasReciente() {
  const ss = SpreadsheetApp.openById(ID_SHEET_NUBE);
  let best = null;
  MESES_NUBE.forEach(function(_,i){
    const sh = ss.getSheetByName(nombreHojaMes(i));
    if (!sh || sh.getLastRow() < 2) return;
    const fecha = sh.getRange(2,2).getValue();
    const d = fecha instanceof Date ? fecha : new Date(fecha);
    if (!best || (!isNaN(d.getTime()) && d > best.fecha)) {
      const payload = leerPayloadMes(i);
      if (payload) best = {fecha:d,payload:payload,periodIndex:i};
    }
  });
  return best;
}

function guardarHistorialPublicacion(snapshotId, fecha, s, payload, chunks) {
  const sh = asegurarHojaPublicaciones();
  const periodo = payload && payload.model && payload.model.month ? payload.model.month : '';
  sh.appendRow([snapshotId,fecha,s.usuario,s.tipo,periodo,String(payload.inputMode||''),chunks]);
  // Conserva las 100 publicaciones más recientes de metadata.
  if (sh.getLastRow() > 101) sh.deleteRows(2, sh.getLastRow() - 101);
}

function subirModeloNube(token, payloadText) {
  const s = exigirSesion(token);
  if (String(s.tipo).toUpperCase() !== 'ADMINISTRADOR') {
    return { ok: false, mensaje: 'Solo un ADMINISTRADOR puede publicar información en la nube.' };
  }
  const payload = JSON.parse(String(payloadText || '{}'));
  if (!payload || !payload.model) return { ok: false, mensaje: 'No se recibió un modelo válido.' };

  const snapshotId = Utilities.getUuid();
  const fecha = new Date().toISOString();
  payload.cloud = { snapshotId: snapshotId, uploadedAt: fecha, uploadedBy: s.usuario };
  const json = JSON.stringify(payload);
  const chunks = [];
  for (let i=0; i<json.length; i+=CHUNK_SIZE) chunks.push(json.slice(i, i+CHUNK_SIZE));

  const periodIndex = indiceMesSeguro(payload.periodIndex);
  if (periodIndex === null) return { ok:false, mensaje:'No fue posible identificar el mes que deseas publicar.' };
  const sh = asegurarHojaMes(periodIndex);
  // Cada mes es un espacio fijo. Una nueva publicación del mismo mes reemplaza solo esa pestaña.
  if (sh.getLastRow() > 1) sh.getRange(2,1,sh.getLastRow()-1,sh.getLastColumn()).clearContent();
  const periodo = payload && payload.model && payload.model.month ? String(payload.model.month) : MESES_NUBE[periodIndex];
  const anio = extraerAnioPayload(payload);
  const rows = chunks.map((c,i)=>[snapshotId,fecha,s.usuario,s.tipo,i+1,chunks.length,periodo,anio,c]);
  sh.getRange(2,1,rows.length,9).setValues(rows);
  guardarHistorialPublicacion(snapshotId, fecha, s, payload, chunks.length);
  // Crea el expediente inicial de cada gerencia a partir del semáforo calculado en el portal.
  if (Array.isArray(payload.workflowSeed)) {
    payload.workflowSeed.forEach(function(x){
      const g = String(x && x.gerencia || '').trim();
      if (g) {
        // Conserva la lectura del semáforo al crear el flujo inicial:
        // Crítico = pendiente prioritario; Atención = ya clasificado en atención.
        const estadoInicial = String(x && x.semaforo || '').toUpperCase() === 'ATENCIÓN' || String(x && x.semaforo || '').toUpperCase() === 'ATENCION'
          ? 'EN_ATENCION'
          : 'PENDIENTE';
        asegurarIncidencia(snapshotId, g, x.prioridad || 'MEDIA', s.usuario, s.tipo, estadoInicial);
      }
    });
  }
  SpreadsheetApp.flush();
  return { ok: true, mensaje: 'Información publicada en la nube.', snapshotId: snapshotId, uploadedAt: fecha, chunks: chunks.length };
}

function leerPayloadNube(periodIndex) {
  const i = indiceMesSeguro(periodIndex);
  if (i !== null) return leerPayloadMes(i);
  const recent = leerPayloadNubeMasReciente();
  if (recent) return recent.payload;

  // Compatibilidad con la versión anterior si todavía existe REPORTIA_NUBE.
  const sh = asegurarHojaNube();
  if (sh.getLastRow() < 2) return null;
  const values = sh.getRange(2,1,sh.getLastRow()-1,7).getValues();
  if (!values.length) return null;
  values.sort((a,b)=>Number(a[4])-Number(b[4]));
  const json = values.map(r=>String(r[6] || '')).join('');
  if (!json) return null;
  return JSON.parse(json);
}


function leerPayloadPorSnapshot(snapshotId) {
  const target = String(snapshotId || '').trim();
  if (!target) return null;
  for (let i=0;i<12;i++) {
    const p = leerPayloadMes(i);
    if (p && p.cloud && String(p.cloud.snapshotId || '') === target) return p;
  }
  return null;
}

function normalizarRol(s) {
  return String(s || '').normalize('NFD').replace(/[\u0300-\u036f]/g,'').toUpperCase().replace(/\./g,'').replace(/\s+/g,' ').trim();
}

function filaPerteneceARol(row, tipo) {
  const rol = normalizarRol(tipo);
  if (rol === 'ADMINISTRADOR') return true;
  const manager = normalizarRol(row && row.manager);
  const aliases = {
    'ADMVA':['ADMVA','ADMINISTRATIVA','ADMINISTRATIVO','ADMINISTRACION'],
    'COMERCIAL':['COMERCIAL'],
    'ESPECIALIZADO':['ESPECIALIZADO'],
    'GENERAL':['GENERAL'],
    'JEF SERV':['JEF SERV','JEFE SERV','JEFATURA SERV'],
    'MANTTO':['MANTTO','MANTENIMIENTO'],
    'OP INTERM':['OP INTERM','OPERACION INTERM','OPERACION INTERMEDIA'],
    'OP PRIMERA':['OP PRIMERA','OPERACION PRIMERA','PRIMERA'],
    'REC HUM':['REC HUM','RECURSOS HUMANOS','RECURSO HUMANO']
  };
  const arr = aliases[rol] || [rol];
  return arr.some(a => manager === a || manager.indexOf(a) >= 0);
}

function filtrarModeloPorRol(model, tipo) {
  if (!model || normalizarRol(tipo) === 'ADMINISTRADOR') return model;
  const copy = JSON.parse(JSON.stringify(model));
  const arrays = ['costOperating','costMaintenance','premises','smo','expenses','expenseSummary','xpv','xpvSummary','costSummary'];
  arrays.forEach(k=>{
    if (Array.isArray(copy[k])) copy[k] = copy[k].filter(r=>filaPerteneceARol(r,tipo));
  });
  // Los acumulados mensuales globales no deben mostrarse a un usuario de área.
  copy.costMonthly = [];
  copy.expenseMonthly = [];
  copy.sourceMode = 'CLOUD';
  return copy;
}

function obtenerMetaNube(token, periodIndex) {
  try {
    exigirSesion(token);
    const p = leerPayloadNube(periodIndex);
    if (!p) return { ok: true, disponible: false, mensaje: 'Aún no hay información publicada por el administrador.' };
    return { ok: true, disponible: true, cloud: p.cloud || {}, month: p.model && p.model.month, periodIndex: p.periodIndex, slot: indiceMesSeguro(periodIndex) };
  } catch (error) { return { ok: false, mensaje: String(error) }; }
}

function obtenerDatosNube(token, periodIndex) {
  try {
    const s = exigirSesion(token);
    const p = leerPayloadNube(periodIndex);
    if (!p) return { ok: true, disponible: false, mensaje: 'Aún no hay información publicada por el administrador.' };
    return {
      ok: true,
      disponible: true,
      cloud: p.cloud || {},
      periodIndex: p.periodIndex,
      inputMode: 'CLOUD',
      model: filtrarModeloPorRol(p.model, s.tipo),
      usuario: s.usuario,
      tipo: s.tipo
    };
  } catch (error) { return { ok: false, mensaje: String(error) }; }
}

function agregarNota(token, snapshotId, gerencia, indicador, texto, parentId, tipoNota) {
  const s = exigirSesion(token);
  const txt = String(texto || '').trim();
  const g = String(gerencia || '').trim();
  if (!g || !txt) return { ok: false, mensaje: 'Gerencia y texto son obligatorios.' };

  const p = leerPayloadPorSnapshot(snapshotId);
  const currentSnapshot = p && p.cloud && p.cloud.snapshotId;
  if (!currentSnapshot || String(snapshotId || '') !== String(currentSnapshot)) {
    return { ok: false, mensaje: 'La información cambió. Actualiza la página antes de registrar la nota.' };
  }
  if (normalizarRol(s.tipo) !== 'ADMINISTRADOR' && !filaPerteneceARol({manager:g}, s.tipo)) {
    return { ok: false, mensaje: 'Tu cuenta no tiene acceso a esta gerencia.' };
  }

  const sh = asegurarHojaNotas();
  const id = Utilities.getUuid();
  const clase = normalizarRol(s.tipo) === 'ADMINISTRADOR' ? 'NOTA_ADMIN' : 'ACCION_CORRECTIVA';
  const tipoFinal = String(tipoNota || clase).toUpperCase();
  sh.appendRow([id,currentSnapshot,g,String(indicador||'SEMÁFORO'),tipoFinal,txt,s.usuario,s.tipo,String(parentId||''),new Date().toISOString(),'ABIERTO']);
  return { ok: true, id: id, mensaje: 'Seguimiento guardado.' };
}

function obtenerNotas(token, gerencia, snapshotId) {
  try {
    const s = exigirSesion(token);
    const g = String(gerencia || '').trim();
    if (normalizarRol(s.tipo) !== 'ADMINISTRADOR' && !filaPerteneceARol({manager:g}, s.tipo)) {
      return { ok: false, mensaje: 'Tu cuenta no tiene acceso a esta gerencia.' };
    }
    const sh = asegurarHojaNotas();
    if (sh.getLastRow() < 2) return { ok: true, notas: [] };
    const vals = sh.getRange(2,1,sh.getLastRow()-1,11).getDisplayValues();
    const out = vals.filter(r => (!snapshotId || r[1] === String(snapshotId)) && (!g || r[2] === g)).map(r=>({

function estadoPermitido(estado) {
  return ['PENDIENTE','EN_ATENCION','POR_VALIDAR','CERRADA'].indexOf(String(estado||'').toUpperCase()) >= 0;
}

function filaIncidencia(sh, snapshotId, gerencia) {
  if (sh.getLastRow() < 2) return 0;
  const vals = sh.getRange(2,1,sh.getLastRow()-1,2).getDisplayValues();
  for (let i=0;i<vals.length;i++) if (vals[i][0] === String(snapshotId) && vals[i][1] === String(gerencia)) return i+2;
  return 0;
}

function asegurarIncidencia(snapshotId, gerencia, prioridad, usuario, tipo, estadoInicial) {
  const sh = asegurarHojaIncidencias();
  let row = filaIncidencia(sh, snapshotId, gerencia);
  if (!row) {
    const now = new Date().toISOString();
    const estado = estadoPermitido(estadoInicial) ? String(estadoInicial).toUpperCase() : 'PENDIENTE';
    sh.appendRow([snapshotId,gerencia,estado,String(prioridad||'MEDIA').toUpperCase(),now,now,usuario||'',tipo||'']);
    row = sh.getLastRow();
  } else if (prioridad) {
    sh.getRange(row,4).setValue(String(prioridad).toUpperCase());
  }
  return row;
}

function sincronizarIncidencias(token, snapshotId, itemsText) {
  const s = exigirSesion(token);
  if (normalizarRol(s.tipo) !== 'ADMINISTRADOR') return {ok:false,mensaje:'Solo el administrador puede sincronizar incidencias.'};
  const p = leerPayloadPorSnapshot(snapshotId);
  const actual = p && p.cloud && p.cloud.snapshotId;
  if (!actual || String(snapshotId||'') !== String(actual)) return {ok:false,mensaje:'La publicación cambió. Actualiza la página.'};
  let items=[];
  try { items=JSON.parse(String(itemsText||'[]')); } catch (_) { return {ok:false,mensaje:'Lista de incidencias no válida.'}; }
  let creadas=0;
  items.forEach(x=>{
    const g=String(x.gerencia||'').trim(); if(!g)return;
    const sh=asegurarHojaIncidencias();
    const existed=filaIncidencia(sh,actual,g);
    asegurarIncidencia(actual,g,x.prioridad||'MEDIA',s.usuario,s.tipo);
    if(!existed)creadas++;
  });
  return {ok:true,creadas:creadas,total:items.length};
}

function actualizarEstadoIncidencia(token, snapshotId, gerencia, estado, prioridad) {
  const s = exigirSesion(token);
  const g = String(gerencia||'').trim();
  const nuevo = String(estado||'').toUpperCase();
  if (!g || !estadoPermitido(nuevo)) return {ok:false,mensaje:'Gerencia o estado no válido.'};
  const p = leerPayloadPorSnapshot(snapshotId);
  const actual = p && p.cloud && p.cloud.snapshotId;
  if (!actual || String(snapshotId||'') !== String(actual)) return {ok:false,mensaje:'La publicación cambió. Actualiza la página.'};
  if (normalizarRol(s.tipo) !== 'ADMINISTRADOR' && !filaPerteneceARol({manager:g}, s.tipo)) return {ok:false,mensaje:'Tu cuenta no tiene acceso a esta gerencia.'};

  const sh = asegurarHojaIncidencias();
  let row = asegurarIncidencia(actual,g,prioridad||'MEDIA',s.usuario,s.tipo);
  const actualEstado = String(sh.getRange(row,3).getDisplayValue()||'PENDIENTE').toUpperCase();
  const admin = normalizarRol(s.tipo) === 'ADMINISTRADOR';
  const permitidasAdmin = ['PENDIENTE','EN_ATENCION','POR_VALIDAR','CERRADA'];
  const permitidasUsuario = ['EN_ATENCION','POR_VALIDAR'];
  if ((admin && permitidasAdmin.indexOf(nuevo)<0) || (!admin && permitidasUsuario.indexOf(nuevo)<0)) return {ok:false,mensaje:'Tu tipo de cuenta no puede aplicar ese cambio de estado.'};
  if (!admin && actualEstado === 'CERRADA') return {ok:false,mensaje:'La incidencia ya está cerrada por el administrador.'};
  sh.getRange(row,3,1,6).setValues([[nuevo,String(prioridad||sh.getRange(row,4).getDisplayValue()||'MEDIA').toUpperCase(),sh.getRange(row,5).getValue()||new Date(),new Date().toISOString(),s.usuario,s.tipo]]);
  return {ok:true,estado:nuevo,mensaje:'Estado actualizado.'};
}
